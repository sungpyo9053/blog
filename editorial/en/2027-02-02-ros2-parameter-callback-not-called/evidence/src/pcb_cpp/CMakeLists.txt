cmake_minimum_required(VERSION 3.8)
project(pcb_cpp)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rcl_interfaces REQUIRED)
add_executable(pcb src/pcb.cpp)
target_compile_options(pcb PRIVATE -Wall -Wextra)
target_link_libraries(pcb rclcpp::rclcpp ${rcl_interfaces_TARGETS})
install(TARGETS pcb DESTINATION lib/${PROJECT_NAME})
ament_package()
