cmake_minimum_required(VERSION 3.10)
project(hzbench)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
add_executable(pub src/pub.cpp)
target_link_libraries(pub PUBLIC rclcpp::rclcpp ${sensor_msgs_TARGETS})
add_executable(count src/count.cpp)
target_link_libraries(count PUBLIC rclcpp::rclcpp ${sensor_msgs_TARGETS})
install(TARGETS pub count DESTINATION lib/${PROJECT_NAME})
ament_package()
